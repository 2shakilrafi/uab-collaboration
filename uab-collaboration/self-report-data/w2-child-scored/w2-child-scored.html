# Function to run GLM model for a given target
run_glm_model <- function(target, data) {
  rec <- recipe(as.formula(paste(target, "~ .")), data = data) %>%
    step_normalize(all_predictors())

  glm_spec <- linear_reg() %>%
    set_engine("lm") %>%
    set_mode("regression")

  wf <- workflow() %>%
    add_model(glm_spec) %>%
    add_recipe(rec)

  folds <- vfold_cv(data, v = 5, repeats = 3)

  res <- fit_resamples(
    wf,
    resamples = folds,
    metrics = metric_set(rmse, rsq),
    control = control_resamples(save_pred = TRUE)
  )

  final_fit <- fit(wf, data = data)

  # Training set R²
  preds_train <- predict(final_fit, data) %>%
    bind_cols(data %>% select(!!target))
  colnames(preds_train) <- c("prediction", "truth")
  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) %>%
    pull(.estimate)

  # Cross-validated R²
  rsq_cv <- collect_metrics(res) %>%
    filter(.metric == "rsq") %>%
    summarise(cv_rsq = mean(mean)) %>%
    pull(cv_rsq)

  # Get coefficients
  coefs <- tidy(final_fit) %>%
    filter(term != "(Intercept)") %>%
    mutate(
      Target = target,
      term = gsub("`", "", term)
    )

  list(
    target = target,
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    coefficients = coefs
  )
}

# Function to run Random Forest model for a given target
run_rf_model <- function(target, data) {
  # Remove rows with missing target values
  data_clean <- data %>% filter(!is.na(.data[[target]]))
  
  # Skip if there are too few rows to model
  if (nrow(data_clean) < 10) {
    message(glue::glue("Skipping {target}: Not enough rows after filtering."))
    return(NULL)
  }

  rec <- recipe(as.formula(paste(target, "~ .")), data = data_clean) %>%
    step_normalize(all_predictors())

  rf_spec <- rand_forest(mtry = tune(), trees = 500, min_n = tune()) %>%
    set_engine("ranger", importance = "permutation") %>%
    set_mode("regression")

  wf <- workflow() %>%
    add_model(rf_spec) %>%
    add_recipe(rec)

  folds <- vfold_cv(data_clean, v = 5, repeats = 3)

  rf_grid <- grid_regular(
    mtry(range = c(2, 10)),
    min_n(range = c(2, 10)),
    levels = 3
  )

  tuned <- tune_grid(
    wf,
    resamples = folds,
    grid = rf_grid,
    metrics = metric_set(rmse, rsq),
    control = control_grid(save_pred = TRUE)
  )

  best_params <- select_best(tuned, metric = "rsq")

  final_wf <- finalize_workflow(wf, best_params)
  final_fit <- fit(final_wf, data = data_clean)

  preds_train <- predict(final_fit, data_clean) %>%
    bind_cols(truth = data_clean[[target]]) %>%
    rename(prediction = .pred)

  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) %>%
    pull(.estimate)

  rsq_cv <- tuned %>%
    collect_metrics() %>%
    filter(.metric == "rsq") %>%
    summarise(cv_rsq = max(mean)) %>%
    pull(cv_rsq)

  vip <- final_fit %>%
    extract_fit_parsnip() %>%
    vip::vi() %>%
    rename(term = Variable, importance = Importance) %>%
    mutate(
      Target = target,
      term = gsub("`", "", term)
    )

  list(
    target = target,
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    importances = vip
  )
}

# Extract all column names as potential targets
targets <- names(df)

# Function to run XGBoost model for a given target
run_xgb_model <- function(target, data) {
  # Remove rows with missing target values
  data_clean <- data %>% filter(!is.na(.data[[target]]))
  
  # Skip if too few rows
  if (nrow(data_clean) < 10) {
    message(glue("Skipping {target}: Not enough rows after filtering."))
    return(NULL)
  }

  rec <- recipe(as.formula(paste(target, "~ .")), data = data_clean) %>%
    step_normalize(all_predictors())

  xgb_spec <- boost_tree(
    trees = tune(),
    tree_depth = tune(),
    learn_rate = tune()
  ) %>%
    set_engine("xgboost") %>%
    set_mode("regression")

  wf <- workflow() %>%
    add_model(xgb_spec) %>%
    add_recipe(rec)

  folds <- vfold_cv(data_clean, v = 5, repeats = 3)

  xgb_grid <- grid_regular(
    trees(range = c(100, 500)),
    tree_depth(range = c(2, 10)),
    learn_rate(range = c(0.01, 0.3)),
    levels = 3
  )

  tuned <- tune_grid(
    wf,
    resamples = folds,
    grid = xgb_grid,
    metrics = metric_set(rmse, rsq),
    control = control_grid(save_pred = TRUE)
  )

  best_params <- select_best(tuned, metric = "rsq")

  final_wf <- finalize_workflow(wf, best_params)
  final_fit <- fit(final_wf, data = data_clean)

  preds_train <- predict(final_fit, data_clean) %>%
    bind_cols(truth = data_clean[[target]]) %>%
    rename(prediction = .pred)

  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) %>%
    pull(.estimate)

  rsq_cv <- tuned %>%
    collect_metrics() %>%
    filter(.metric == "rsq") %>%
    summarise(cv_rsq = max(mean)) %>%
    pull(cv_rsq)

  vip <- final_fit %>%
    extract_fit_parsnip() %>%
    vip::vi() %>%
    rename(term = Variable, importance = Importance) %>%
    mutate(
      Target = target,
      term = gsub("`", "", term)
    )

  list(
    target = target,
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    importances = vip
  )
}

# Run model for each target
results_xgb <- map(targets, ~ run_xgb_model(.x, df)) %>%
  compact()  # Remove NULL results

# Define pastel palette
pastel_palette <- brewer.pal(8, "Pastel2")

# Plot each target’s importances individually
walk(results_xgb, function(res) {
  n_features <- nrow(res$importances)
  colors <- rep(pastel_palette, length.out = n_features)

  p <- ggplot(res$importances, aes(x = reorder(term, importance), y = importance)) +
    geom_col(fill = colors) +
    coord_flip() +
    theme_minimal(base_size = 14) +
    labs(
      title = paste("XGBoost Feature Importances for:", res$target),
      subtitle = paste0("R² (Train): ", round(res$rsq_train, 3),
                        " | R² (CV): ", round(res$rsq_cv, 3)),
      x = "Features", y = "Importance (Gain)"
    )

  print(p)
})
