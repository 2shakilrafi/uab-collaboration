results <- lapply(names(psychometrics), function(var) {
  values <- data[[var]]
  values <- values[!is.na(values)]
  if (length(values) >= 3 && length(values) <= 5000) {
    test <- shapiro.test(values)
    tibble(
      Variable = psychometrics[[var]],
      W_statistic = test$statistic,
      p_value = test$p.value
    )
  } else {
    tibble(
      Variable = psychometrics[[var]],
      W_statistic = NA,
      p_value = NA
    )
  }
})
