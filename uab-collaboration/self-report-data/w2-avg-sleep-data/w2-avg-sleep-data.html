# Function to build and tune model for a given target
run_model <- function(target, data) {
  rec <- recipe(as.formula(paste(target, "~ .")), data = data) %>%
    step_normalize(all_predictors())

  xgb_spec <- boost_tree(
    trees = tune(),
    tree_depth = tune(),
    learn_rate = tune(),
    loss_reduction = tune(),
    sample_size = tune(),
    mtry = tune()
  ) %>%
    set_engine("xgboost") %>%
    set_mode("regression")

  wf <- workflow() %>%
    add_model(xgb_spec) %>%
    add_recipe(rec)

  folds <- vfold_cv(data, v = 5)

  grid_vals <- grid_latin_hypercube(
    trees(),
    tree_depth(),
    learn_rate(),
    loss_reduction(),
    sample_size = sample_prop(),
    finalize(mtry(), data %>% select(-all_of(target))),
    size = 15
  )

  tuned <- tune_grid(
    wf,
    resamples = folds,
    grid = grid_vals,
    metrics = metric_set(rmse, rsq),
    control = control_grid(save_pred = TRUE)
  )

  best <- select_best(tuned, metric = "rmse")

  final_wf <- finalize_workflow(wf, best)
  final_fit <- fit(final_wf, data = data)

  # Training set R²
  preds_train <- predict(final_fit, data) %>%
    bind_cols(data %>% select(!!target))
  colnames(preds_train) <- c("prediction", "truth")
  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) %>%
    pull(.estimate)

  # Cross-validated R²
  best_config <- best %>% select(-.config)
  tuned_metrics <- collect_predictions(tuned) %>%
    inner_join(best_config, by = colnames(best_config)) %>%
    group_by(id) %>%
    rsq(truth = !!sym(target), estimate = .pred) %>%
    summarise(cv_rsq = mean(.estimate))

  rsq_cv <- mean(tuned_metrics$cv_rsq)

  # Feature importance (normalized gain)
  imp <- final_fit %>%
    pull_workflow_fit() %>%
    vip::vi(type = "gain") %>%
    mutate(
      NormalizedGain = Importance / sum(Importance),
      Target = target
    )

  list(
    target = target,
    rmse = show_best(tuned, metric = "rmse", n = 1),
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    workflow = final_wf,
    fit = final_fit,
    importance = imp
  )
}

# Function to run GLM model for a given target
run_glm_model <- function(target, data) {
  rec <- recipe(as.formula(paste(target, "~ .")), data = data) %>%
    step_normalize(all_predictors())

  glm_spec <- linear_reg() %>%
    set_engine("lm") %>%
    set_mode("regression")

  wf <- workflow() %>%
    add_model(glm_spec) %>%
    add_recipe(rec)

  folds <- vfold_cv(data, v = 5, repeats = 3)

  res <- fit_resamples(
    wf,
    resamples = folds,
    metrics = metric_set(rmse, rsq),
    control = control_resamples(save_pred = TRUE)
  )

  final_fit <- fit(wf, data = data)

  # Training set R²
  preds_train <- predict(final_fit, data) %>%
    bind_cols(data %>% select(!!target))
  colnames(preds_train) <- c("prediction", "truth")
  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) %>%
    pull(.estimate)

  # Cross-validated R²
  rsq_cv <- collect_metrics(res) %>%
    filter(.metric == "rsq") %>%
    summarise(cv_rsq = mean(mean)) %>%
    pull(cv_rsq)

  # Get coefficients
  coefs <- tidy(final_fit) %>%
    filter(term != "(Intercept)") %>%
    mutate(
      Target = target,
      term = gsub("`", "", term)  # Clean up variable names
    )

  list(
    target = target,
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    workflow = wf,
    fit = final_fit,
    coefficients = coefs
  )
}

  mutate(cluster = umap_df$cluster)

# Get summary stats per cluster
cluster_summary <- df_clustered %>%
  pivot_longer(-cluster, names_to = "variable", values_to = "value") %>%
  group_by(cluster, variable) %>%
  summarise(
    mean = mean(value),
    sd = sd(value),
    .groups = "drop"
  ) %>%
  arrange(cluster, variable)

# Optional: pivot wider to see everything in a big table
cluster_summary_wide <- cluster_summary %>%
  pivot_wider(
    names_from = variable,
    values_from = c(mean, sd),
    names_glue = "{variable}_{.value}"
  )

# View the summary
print(cluster_summary_wide, n = Inf)

library(tidyverse)

# Combine cluster labels with normalized data
df_clustered <- df_norm %>%
  mutate(cluster = umap_df$cluster)

# Get variable names (excluding the cluster label)
vars <- names(df_clustered)[names(df_clustered) != "cluster"]

# Loop through variables and generate violin plots
for (v in vars) {
  p <- ggplot(df_clustered, aes(x = cluster, y = .data[[v]], fill = cluster)) +
    geom_violin(trim = FALSE, alpha = 0.7, color = NA) +
    geom_jitter(width = 0.15, size = 1.2, alpha = 0.4, show.legend = FALSE) +
    scale_fill_brewer(palette = "Pastel1") +
    theme_minimal(base_size = 14) +
    labs(
      title = paste("Violin Plot of", v, "by Cluster"),
      x = "Cluster",
      y = paste("Standardized", v)
    )

  print(p)
}
