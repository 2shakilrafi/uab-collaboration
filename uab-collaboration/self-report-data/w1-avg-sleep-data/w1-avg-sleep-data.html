  }

library(tidyverse)

# Read data
df <- read.csv("W1_AvgSleepData.csv")

# Remove SID and c1remove
df <- df %>% select(-SID, -c1remove)

# Drop rows with any NA values
df_clean <- df %>% drop_na()

# Scale all columns
scaled_df <- scale(df_clean) %>% as.data.frame()

# Initialize result storage
results <- list()

# Loop through each column as target
for (target in colnames(scaled_df)) {
  
  predictors <- setdiff(colnames(scaled_df), target)
  
  # Create formula
  formula <- as.formula(paste(target, "~", paste(predictors, collapse = "+")))
  
  # Fit random forest
  rf_model <- randomForest(formula, data = scaled_df, importance = TRUE)
  
  # Get R-squared and importance
  rsq <- rf_model$rsq[length(rf_model$rsq)]
  importance_vals <- importance(rf_model)
  
  results[[target]] <- list(
    target = target,
    predictors = predictors,
    r_squared = rsq,
    importance = importance_vals
  )
}

# Summarize R-squared values
r_squared_summary <- tibble(
  Target = names(results),
  R_squared = sapply(results, function(x) x$r_squared)
)

# Display sorted by R² descending
print(r_squared_summary %>% arrange(desc(R_squared)))

# Create a folder to save plots (optional)

# Loop through each result and plot importance
for (res in results) {
  
  # Get importance data
  imp_df <- as.data.frame(res$importance)
  imp_df$Predictor <- rownames(imp_df)
  imp_df <- imp_df %>% 
    rename(Importance = `%IncMSE`) %>% 
    arrange(desc(Importance))
  
  # Create plot
  p <- ggplot(imp_df, aes(x = reorder(Predictor, Importance), y = Importance, fill = Importance)) +
    geom_bar(stat = "identity") +
    coord_flip() +
    scale_fill_gradient(low = "skyblue", high = "darkblue") +
    labs(title = paste("Feature Importance for Predicting", res$target),
         x = "Predictor",
         y = "% Increase in MSE when Removed") +
    theme_minimal(base_size = 12) +
    theme(legend.position = "none",
          axis.text.y = element_text(size = 10))
  
  # Print the plot (if in interactive mode)
  print(p)
  
  # Save each plot to disk
  ggsave(filename = paste0("feature_importance_plots/", res$target, "_importance.png"),
         plot = p, width = 8, height = 5, dpi = 300)
}
