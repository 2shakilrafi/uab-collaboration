    mtry = tune()
  ) |>
    set_engine("xgboost") |>
    set_mode("regression")

  wf <- workflow() |>
    add_model(xgb_spec) |>
    add_recipe(rec)

  folds <- vfold_cv(data, v = 5)

  grid_vals <- grid_latin_hypercube(
    trees(),
    tree_depth(),
    learn_rate(),
    loss_reduction(),
    sample_size = sample_prop(),
    finalize(mtry(), data %>% select(-all_of(target))),
    size = 15
  )

  tuned <- tune_grid(
    wf,
    resamples = folds,
    grid = grid_vals,
    metrics = metric_set(rmse, rsq),
    control = control_grid(save_pred = TRUE)
  )

  best <- select_best(tuned, metric = "rmse")

  final_wf <- finalize_workflow(wf, best)
  final_fit <- fit(final_wf, data = data)

  preds_train <- predict(final_fit, data) |>
    bind_cols(data %>% select(!!target))
  colnames(preds_train) <- c("prediction", "truth")
  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) |> pull(.estimate)

  best_config <- best %>% select(-.config)
  tuned_metrics <- collect_predictions(tuned) |>
    inner_join(best_config, by = colnames(best_config)) |>
    group_by(id) |>
    rsq(truth = !!sym(target), estimate = .pred) |>
    summarise(cv_rsq = mean(.estimate))

  rsq_cv <- mean(tuned_metrics$cv_rsq)

  imp <- final_fit |>
    pull_workflow_fit() |>
    vip::vi(type = "gain") |>
    mutate(
      NormalizedGain = Importance / sum(Importance),
      Target = target
    )

  list(
    target = target,
    rmse = show_best(tuned, metric = "rmse", n = 1),
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    workflow = final_wf,
    fit = final_fit,
    importance = imp
  )
}

run_glm_model <- function(target, data, color) {
  rec <- recipe(as.formula(paste(target, "~ .")), data = data) |>
    step_impute_median(all_predictors()) |>
    step_normalize(all_predictors())

  glm_spec <- linear_reg() |>
    set_engine("lm")

  wf <- workflow() |>
    add_model(glm_spec) |>
    add_recipe(rec)

  folds <- vfold_cv(data, v = 5)

  # Fit model on full data
  final_fit <- fit(wf, data)

  # R² on training data
  preds_train <- predict(final_fit, data) |>
    bind_cols(data %>% select(!!target))
  colnames(preds_train) <- c("prediction", "truth")
  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) |> pull(.estimate)

  # R² via cross-validation
  cv_preds <- fit_resamples(
    wf,
    resamples = folds,
    metrics = metric_set(rsq)
  )
  rsq_cv <- collect_metrics(cv_preds) |>
    filter(.metric == "rsq") |>
    summarise(mean = mean(mean)) |>
    pull(mean)

run_rf_model <- function(target, data) {
  rec <- recipe(as.formula(paste(target, "~ .")), data = data) |>
    step_impute_median(all_predictors()) |>  # Handle NAs
    step_normalize(all_predictors())

  rf_spec <- rand_forest(
    mtry = tune(),
    min_n = tune(),
    trees = 500
  ) |>
    set_engine("ranger", importance = "impurity") |>
    set_mode("regression")

  wf <- workflow() |>
    add_model(rf_spec) |>
    add_recipe(rec)

  folds <- vfold_cv(data, v = 5)

  grid_vals <- grid_latin_hypercube(
    mtry(range = c(1, ncol(data) - 1)),
    min_n(),
    size = 10
  )

  tuned <- tune_grid(
    wf,
    resamples = folds,
    grid = grid_vals,
    metrics = metric_set(rmse, rsq),
    control = control_grid(save_pred = TRUE)
  )

  best <- select_best(tuned, metric = "rmse")
  final_wf <- finalize_workflow(wf, best)
  final_fit <- fit(final_wf, data = data)

  preds_train <- predict(final_fit, data) |>
    bind_cols(data %>% select(!!target))
  colnames(preds_train) <- c("prediction", "truth")
  rsq_train <- rsq(preds_train, truth = truth, estimate = prediction) |> pull(.estimate)

  best_config <- best %>% select(-.config)
  tuned_metrics <- collect_predictions(tuned) |>
    inner_join(best_config, by = colnames(best_config)) |>
    group_by(id) |>
    rsq(truth = !!sym(target), estimate = .pred) |>
    summarise(cv_rsq = mean(.estimate))

  rsq_cv <- mean(tuned_metrics$cv_rsq)

  imp <- final_fit |>
    pull_workflow_fit() |>
    vip::vi() |>
    mutate(
      NormalizedGain = Importance / sum(Importance),
      Target = target
    )

  list(
    target = target,
    rmse = show_best(tuned, metric = "rmse", n = 1),
    rsq_train = rsq_train,
    rsq_cv = rsq_cv,
    workflow = final_wf,
    fit = final_fit,
    importance = imp
  )
}
